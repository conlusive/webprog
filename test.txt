# Команди, якими перевірено сервер (curl).
# Запуск сервера (у першому терміналі):
node main.js -h 127.0.0.1 -p 3000 -c ./cache

# --- Запуск без обов'язкової опції (очікуємо повідомлення і код 1) ---
node main.js -h 127.0.0.1 -p 3000
echo "код завершення: $?"

# --- Недозволені методи: очікуємо 405 і заголовок Allow ---
curl -i -X POST http://127.0.0.1:3000/404
curl -I http://127.0.0.1:3000/404   # HEAD

# --- Перевірка шляху: очікуємо 404 ---
curl -i http://127.0.0.1:3000/abc
curl -i http://127.0.0.1:3000/
curl -i http://127.0.0.1:3000/4040
curl -i --path-as-is http://127.0.0.1:3000/../etc/passwd
